#!/usr/bin/env python3
"""Forge server 1.7.10 без installer --installServer (docs/server-vm.md).

  forge.py DIR     -> DIR/forge-...-universal.jar, DIR/minecraft_server.1.7.10.jar, DIR/libraries/...

Штатный установщик 1.7.10 ходит на http://files.minecraftforge.net (редирект на https Java 8 не проходит) и на
мёртвые адреса Mojang — поэтому разбираем install_profile.json сами, качаем по https с запасными зеркалами и
сверяем Class-Path универсального jar: чего нет — FAIL.
"""
import json, os, sys, urllib.request, zipfile, io

MC, FV = "1.7.10", "1.7.10-10.13.4.1614-1.7.10"
INSTALLER = f"https://maven.minecraftforge.net/net/minecraftforge/forge/{FV}/forge-{FV}-installer.jar"
MIRRORS = ["https://maven.minecraftforge.net/", "https://libraries.minecraft.net/", "https://repo1.maven.org/maven2/"]
UA = {"User-Agent": "zavod-server-setup"}


def get(url):
    with urllib.request.urlopen(urllib.request.Request(url, headers=UA), timeout=120) as r:
        return r.read()


def lib_path(name):
    g, a, v = name.split(":")[:3]
    return f"{g.replace('.', '/')}/{a}/{v}/{a}-{v}.jar"


def fetch_lib(name, url, dest):
    if os.path.isfile(dest) and os.path.getsize(dest) > 0:
        return "есть"
    rel = lib_path(name)
    bases = []
    if url:
        bases.append(url.replace("http://files.minecraftforge.net/maven/", "https://maven.minecraftforge.net/")
                        .replace("http://", "https://"))
    bases += [m for m in MIRRORS if m not in bases]
    err = None
    for b in bases:
        try:
            data = get(b.rstrip("/") + "/" + rel)
            os.makedirs(os.path.dirname(dest), exist_ok=True)
            open(dest, "wb").write(data)
            return b
        except Exception as e:
            err = e
    raise SystemExit(f"FAIL библиотека {name}: {err}")


def main(d):
    os.makedirs(d, exist_ok=True)
    z = zipfile.ZipFile(io.BytesIO(get(INSTALLER)))
    prof = json.loads(z.read("install_profile.json"))
    uni = prof["install"]["filePath"]
    open(os.path.join(d, uni), "wb").write(z.read(uni))
    n = 0
    for lib in prof["versionInfo"]["libraries"]:
        if not lib.get("serverreq") or lib["name"].startswith("net.minecraftforge:forge:"):
            continue
        fetch_lib(lib["name"], lib.get("url"), os.path.join(d, "libraries", lib_path(lib["name"])))
        n += 1
    srv = os.path.join(d, f"minecraft_server.{MC}.jar")
    if not os.path.isfile(srv):
        man = json.loads(get("https://piston-meta.mojang.com/mc/game/version_manifest_v2.json"))
        vurl = next(v["url"] for v in man["versions"] if v["id"] == MC)
        open(srv, "wb").write(get(json.loads(get(vurl))["downloads"]["server"]["url"]))
    mf = zipfile.ZipFile(os.path.join(d, uni)).read("META-INF/MANIFEST.MF").decode().replace("\r\n ", "").replace("\n ", "")
    cp = next((l.split(":", 1)[1].split() for l in mf.splitlines() if l.startswith("Class-Path:")), [])
    miss = [p for p in cp if not os.path.isfile(os.path.join(d, p))]
    if miss:
        raise SystemExit("FAIL нет в Class-Path: " + " ".join(miss))
    print(f"forge OK: {uni}, библиотек {n}, Class-Path {len(cp)} — всё на месте")
    return uni


if __name__ == "__main__":
    if len(sys.argv) != 2:
        sys.exit(__doc__)
    main(sys.argv[1])
