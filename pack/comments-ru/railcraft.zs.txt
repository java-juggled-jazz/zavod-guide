// ====================================================================================
//                 БАЛАНС КОКСОВЫХ ПЕЧЕЙ RAILCRAFT
// В railcraft.cfg НЕТ настроек времени коксования и выхода креозота — файл railcraft.json
// Railcraft просто не читает. Меняем рецепты через ModTweaker.
// По умолчанию в Railcraft 9.x: уголь -> кокс за 1800 тиков (90 с) + 500 mB креозота;
// бревно -> древесный уголь за 1800 тиков + 250 mB.
// ====================================================================================

val creosote = <liquid:creosote>;

// Каменный уголь -> кокс: 1800 тиков (90 с), креозот 500 -> 250 mB (ценный дозированный продукт)
mods.railcraft.CokeOven.removeRecipe(<Railcraft:fuel.coke>);
mods.railcraft.CokeOven.addRecipe(<minecraft:coal>, true, false, <Railcraft:fuel.coke>, creosote * 250, 1800);

// Угольный блок -> блок кокса (x9 по времени и креозоту, как в оригинале)
mods.railcraft.CokeOven.removeRecipe(<Railcraft:cube:0>);
mods.railcraft.CokeOven.addRecipe(<minecraft:coal_block>, false, false, <Railcraft:cube:0>, creosote * 2250, 5400);

// Бревно -> древесный уголь: креозот 250 -> 125 mB (у древесины меньше смолы)
mods.railcraft.CokeOven.removeRecipe(<minecraft:coal:1>);
for log in <ore:logWood>.items {
    mods.railcraft.CokeOven.addRecipe(log, false, false, <minecraft:coal:1>, creosote * 125, 1800);
}
