// ====================================================================================
// ДРАГОЦЕННЫЕ МЕТАЛЛЫ: КИСЛОТНЫЙ АФФИНАЖ
// Кислоты - из химцеха (chem.zs, чан EnderIO): золото - азотная кислота, платиноиды - царская водка.
// (аффинаж требует чистых реактивов: соли из обычной воды загрязняют металл).
//   Золото: азотная кислота растворяет всё, кроме золота ("разделение кислотой").
//   Платиноиды: серная кислота выщелачивает никель и медь из анодного шлама, платиноиды остаются
//     концентратом (так работают никелевые заводы). Концентрат = платиновая пыль TF.
//   Иридий: тяжёлая нерастворимая фракция концентрата - отделяется в термоцентрифуге IC2
//     (9 пыли концентрата -> 8 платиновой пыли + 1 осколок иридия (1/9 руды) + самородок золота),
//     config/ic2/thermal_centrifuge.ini.
// Анодный шлам (zavod:anode_slime) - ТОЛЬКО от электролитического рафинирования меди и никеля (ниже).
// Богатый шлак TE больше не источник платины: он падает с любой руды, а платиноиды бывают только в медно-никелевых.
// ====================================================================================

val saltpeter = <Railcraft:dust:2>;
val anodeSlime = <zavod:anode_slime>;               // анодный шлам (мод Zavod)

// --- Золото: 3 дроблёной золотой руды + селитра + 1000 mB воды -> 4 золотой пыли ---
// (ванильный путь IC2 через мойку и центрифугу даёт примерно 2.2 пыли с руды, здесь 2.67 - но тратится селитра)
mods.forestry.Carpenter.addRecipe(<ThermalFoundation:material:1> * 4, [
    [<ore:crushedGold>, <ore:crushedGold>, <ore:crushedGold>]
], <liquid:nitricacid> * 250, 80);

// --- Платиноиды: 2 анодного шлама + сера + 1000 mB воды -> 1 пыль платинового концентрата ---
mods.forestry.Carpenter.addRecipe(<ThermalFoundation:material:37>, [
    [anodeSlime, anodeSlime]
], <liquid:aquaregia> * 500, 80);

// --- Электролитическое рафинирование (индукционная плавильня = электролизная ванна) ---
// Медный купорос: медь + серная кислота (сера + вода, кислота делается на месте) -> электролит.
// (купорос теперь из химцеха: чан EnderIO + кристаллизация, chem.zs)
// Медь: 2 медной пыли + купорос (подпитка электролита) -> 2 слитка + 10% анодного шлама
mods.thermalexpansion.Smelter.addRecipe(4000, <ThermalFoundation:material:32> * 2, <zavod:copper_vitriol>, <ThermalFoundation:material:64> * 2, anodeSlime, 10);
// Никель: 2 никелевой пыли + сера (сульфатный электролит) -> 2 слитка + 15% анодного шлама (норильская схема)
mods.thermalexpansion.Smelter.addRecipe(4000, <ThermalFoundation:material:36> * 2, <ThermalFoundation:material:16>, <ThermalFoundation:material:68> * 2, anodeSlime, 15);

// ====================================================================================
// ШЛАК: ОДИН МАТЕРИАЛ ИЗ ЛЮБОЙ МЕТАЛЛУРГИИ
// Шлак IC2 (доменная печь) и шлак TE (индукционная плавильня) - одно и то же: словарь crystalSlag.
// Центрифугирование шлака в золото и угольную пыль убрано (config/ic2/thermal_centrifuge.ini).
// Применение как в реальности: шлакоцемент и томасшлак (фосфорное удобрение).
// Шлаковата: шлак TE в красной печи TE -> минеральная вата (так в TE изначально).
// ====================================================================================
<ore:crystalSlag>.add(<IC2:itemSlag>);
// Шлакоцемент -> строительная пена IC2 (вместо 6 каменной пыли)
recipes.addShapeless(<IC2:itemPartCFPowder> * 2, [<ore:crystalSlag>, <ore:crystalSlag>, <minecraft:sand>, <minecraft:clay_ball>]);
// Томасшлак: молотый шлак -> удобрение IC2
mods.ic2.Macerator.addRecipe(<IC2:itemFertilizer> * 2, <IC2:itemSlag>);

// ====================================================================================
// ПЛАТИНА В ТЕХНИКЕ (реальные применения)
//   катализатор риформинга -> нефтеперегонный завод BC;
//   свечи зажигания -> ДВС BC (global.zs);
//   фильеры для стекловолокна -> стекловолоконный кабель IC2 (ниже);
//   энддериум TE (резонансные ячейки) - и так на платине.
// ====================================================================================
recipes.remove(<BuildCraft|Factory:refineryBlock>);
recipes.addShaped(<BuildCraft|Factory:refineryBlock>, [
    [<ore:plateSteel>,     <ore:ingotPlatinum>, <ore:plateSteel>],
    [<BuildCraft|Factory:tankBlock>, <ore:gearDiamond>, <BuildCraft|Factory:tankBlock>],
    [<ore:plateSteel>,     <IC2:itemRecipePart:5>, <ore:plateSteel>]
]);

// --- Стекловолоконный кабель IC2: только термоионный фабрикатор Forestry ---
// Расплавленное стекло тянут через платиновую фильеру (слиток платины - шаблон, НЕ расходуется).
// Вручную больше не крафтится. 1000 mB стекла + 2 энергиевой пыли + серебряная пыль -> 2 кабеля.
recipes.remove(<IC2:itemCable:9>);
mods.forestry.ThermionicFabricator.addCast(<IC2:itemCable:9> * 2, [
    [<IC2:itemDust2:2>, <ore:dustSilver>, <IC2:itemDust2:2>]
], 1000, <ThermalFoundation:material:69>);
