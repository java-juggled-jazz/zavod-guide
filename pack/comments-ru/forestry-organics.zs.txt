// ====================================================================================
// ORGANICS FORESTRY -> ПРОМЫШЛЕННОСТЬ
// ПОЧЕМУ: Forestry - мод про производство органики, а его продукты почти никуда не шли.
// ТЕПЕРЬ: торф -> кокс, воск и бумага -> изоляция кабелей, компост -> биогаз,
//   фрукты и мёд -> консервы, прополис -> клей, сложное удобрение -> удобрение IC2.
// Уже работает без скрипта: жидкотопливный генератор тепла IC2 жжёт биомассу и биоэтанол Forestry.
// ====================================================================================

// --- Торфяной кокс: битуминозный торф в коксовой печи Railcraft (возобновляемый кокс) ---
mods.railcraft.CokeOven.addRecipe(<Forestry:bituminousPeat>, false, false, <Railcraft:fuel.coke>, <liquid:creosote> * 150, 1800);

// --- Бумажно-восковая изоляция (как у ранних кабелей) - только низкое напряжение ---
// Медь и олово: голый кабель + бумага + пчелиный воск. Золото и высоковольтный - только резина.
// Резиновая изоляция - ТОЛЬКО плотник (оболочку выдавливают на станке): 3 медных/оловянных + 1 резина -> 3;
//   2 золотых + 2 резины -> 2; высоковольтный + 2 резины -> 1. Вручную резиной не изолировать.
// Вручную - только низкое напряжение (медь, олово): бумага + пчелиный воск -> 1, или 4 кабеля + шерсть -> 4 (оплётка).
recipes.remove(<IC2:itemCable>);
recipes.remove(<IC2:itemCable:13>);
recipes.remove(<IC2:itemCable:3>);
recipes.remove(<IC2:itemCable:6>);
recipes.addShapeless(<IC2:itemCable>,    [<IC2:itemCable:1>,  <minecraft:paper>, <Forestry:beeswax>]);
recipes.addShapeless(<IC2:itemCable:13>, [<IC2:itemCable:10>, <minecraft:paper>, <Forestry:beeswax>]);
recipes.addShapeless(<IC2:itemCable> * 4,    [<IC2:itemCable:1>,  <IC2:itemCable:1>,  <IC2:itemCable:1>,  <IC2:itemCable:1>,  <minecraft:wool:*>]);
recipes.addShapeless(<IC2:itemCable:13> * 4, [<IC2:itemCable:10>, <IC2:itemCable:10>, <IC2:itemCable:10>, <IC2:itemCable:10>, <minecraft:wool:*>]);
mods.forestry.Carpenter.addRecipe(<IC2:itemCable> * 3,    [[<IC2:itemCable:1>,  <IC2:itemCable:1>,  <IC2:itemCable:1>],  [null, <ore:itemRubber>, null]], 20);
mods.forestry.Carpenter.addRecipe(<IC2:itemCable:13> * 3, [[<IC2:itemCable:10>, <IC2:itemCable:10>, <IC2:itemCable:10>], [null, <ore:itemRubber>, null]], 20);
mods.forestry.Carpenter.addRecipe(<IC2:itemCable:3> * 2,  [[<IC2:itemCable:2>,  <IC2:itemCable:2>], [<ore:itemRubber>, <ore:itemRubber>]], 30);
mods.forestry.Carpenter.addRecipe(<IC2:itemCable:6>,      [[<ore:itemRubber>, <IC2:itemCable:5>, <ore:itemRubber>]], 40);

// --- Компост Forestry -> биомасса IC2 (для ферментера) ---
mods.ic2.Canner.addEnrichRecipe(<liquid:ic2biomass> * 1000, <liquid:water> * 1000, <Forestry:fertilizerBio> * 2);

// --- Консервы: фрукты и мёд Forestry в жестяные банки IC2 (еда для шлемов нано/квант) ---
mods.ic2.Canner.addBottleRecipe(<IC2:itemTinCanFilled> * 2, <IC2:itemTinCan> * 2, <Forestry:fruits:*>);
mods.ic2.Canner.addBottleRecipe(<IC2:itemTinCanFilled> * 4, <IC2:itemTinCan> * 4, <Forestry:honeyedSlice>);
mods.ic2.Canner.addBottleRecipe(<IC2:itemTinCanFilled>,     <IC2:itemTinCan>,     <Forestry:honeyDrop>);

// --- Прополис - пчелиный клей: второй путь к липкому поршню (первый - латекс, rubber.zs) ---
recipes.addShapeless(<minecraft:sticky_piston>, [<minecraft:piston>, <Forestry:propolis:*>]);

// --- Сложное удобрение Forestry -> удобрение IC2 (для Crop-Matron) ---
recipes.addShapeless(<IC2:itemFertilizer> * 2, [<Forestry:fertilizerCompound>, <Forestry:fertilizerCompound>]);

// ====================================================================================
// ПРОМЫШЛЕННАЯ ОРГАНИКА: КУДА ДЕВАТЬ УРОЖАЙ

mods.ic2.Canner.addEnrichRecipe(<liquid:ic2biomass> * 1000, <liquid:water> * 1000, <minecraft:carrot> * 8);

    [<minecraft:paper>,    <minecraft:paper>, <minecraft:paper>],

    [<minecraft:paper>,    <minecraft:paper>, <minecraft:paper>]

// ДОБЫЧА С МОНСТРОВ -> ХИМИЯ
// Суперфосфат (как у Лоуса в 1842): костная мука + серная кислота (сера) + вода -> 4 сложных удобрения Forestry.

// ====================================================================================
// ЛЕСОХИМИЯ: ЛИСТВА И ХВОЯ (листья собираются ножницами)
//   хвоя (листва ели) -> соковыжималка: 8 штук -> 100 mB масла (хвойное масло; как растительное: олифа, лак, пропитка)
//     + 30% мульчи. Лаком на этом масле покрывают стальные гильзы.
//   любая листва -> ферментер Forestry -> биомасса (хвоя бродит хуже: смолы мешают брожению)
//   (уже было в IC2: 8 листьев в дробителе -> биоплевелы -> биомасса)
// Дёготь из бересты = креозот: берёза в коксовой печи и так даёт уголь + смолу.
// ====================================================================================

mods.forestry.Fermenter.addRecipe(<liquid:biomass>, <minecraft:leaves:3>, <liquid:water>, 100, 1.0);
mods.forestry.Fermenter.addRecipe(<liquid:biomass>, <minecraft:leaves2:0>, <liquid:water>, 100, 1.0);
mods.forestry.Fermenter.addRecipe(<liquid:biomass>, <minecraft:leaves2:1>, <liquid:water>, 100, 1.0);
