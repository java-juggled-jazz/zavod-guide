// ====================================================================================
// ДЕРЕВООБРАБОТКА: РУКАМИ ПЛОХО, СТАНКОМ ХОРОШО
// ПОЧЕМУ: топором из бревна 4 доски - это сказка. Ручной распил - половина в стружку.
// ТЕПЕРЬ: вручную бревно -> 2 доски, 2 доски -> 2 палки.
//   Пилорама TE: бревно -> 6 досок + опилки (её рецепты строятся из исходных, проверь в NEI).
//   Блокорезка IC2: бревно -> 8 досок (тонкий пропил), 2 доски -> 6 палок (config/ic2/block_cutter.ini).
// Опилки: древесная масса для бумаги и бездымный порох (нитроцеллюлоза).
// Деревья Extra Trees - после jar Binnie.
// ====================================================================================

val sawdust = <ThermalExpansion:material:512>;

// --- Ванильные доски: 2 с бревна ---
recipes.remove(<minecraft:planks:*>);
recipes.addShapeless(<minecraft:planks:0> * 2, [<minecraft:log:0>]);
recipes.addShapeless(<minecraft:planks:1> * 2, [<minecraft:log:1>]);
recipes.addShapeless(<minecraft:planks:2> * 2, [<minecraft:log:2>]);
recipes.addShapeless(<minecraft:planks:3> * 2, [<minecraft:log:3>]);
recipes.addShapeless(<minecraft:planks:4> * 2, [<minecraft:log2:0>]);
recipes.addShapeless(<minecraft:planks:5> * 2, [<minecraft:log2:1>]);

// --- Палки: 2 доски -> 2 палки (было 4) ---
recipes.remove(<minecraft:stick>);
recipes.addShaped(<minecraft:stick> * 2, [[<ore:plankWood>], [<ore:plankWood>]]);

// ====================================================================================
// ОПИЛКИ
// ====================================================================================
// Древесная масса Forestry (-> бумага, картон): опилки, разваренные в воде
mods.forestry.Carpenter.addRecipe(<Forestry:woodPulp> * 2, [
    [sawdust, sawdust]
], <liquid:water> * 250, 20);
// Бездымный порох (нитроцеллюлоза): богатую целлюлозой стружку нитруют смесью кислот.
// Азотная кислота - из селитры, серная - из серы. Нитроцеллюлозу желатинируют спиртом (пироксилиновый порох),
// поэтому нужен биоэтанол (БЫЛО: вода).
// Нитрование идёт только с охлаждением (ледяная баня), иначе реакция уходит вразнос: 1 лёд расходуется.
// Лёд любой: натуральный (шёлковое касание) или из Glacial Precipitator TE.
// Сильнее дымного пороха Railcraft (2 селитры + сера + уголь -> 2): здесь те же 2 селитры дают 4 пороха.
// ТЕПЕРЬ: две стадии и свой предмет (zavod:powder_smokeless), а не ванильный порох - бездымный идёт только в патроны.
// 1) нитрование (плотник, лёд, вода на промывку) -> 4 нитроцеллюлозы; 2) желатинизация спиртом -> 4 бездымного пороха.
mods.forestry.Carpenter.addRecipe(<zavod:nitrocellulose> * 4, [
    [sawdust, sawdust,         sawdust],
    [null,    <minecraft:ice>, null]
], <liquid:nitricacid> * 500, 60);
mods.forestry.Carpenter.addRecipe(<zavod:powder_smokeless> * 4, [
    [<zavod:nitrocellulose>, <zavod:nitrocellulose>],
    [<zavod:nitrocellulose>, <zavod:nitrocellulose>]
], <liquid:bioethanol> * 250, 40);
