// ====================================================================================
// НЕФТЕХИМИЯ (нефть BuildCraft как сырьё, а не только топливо)
// Рецепт нефтеперегонного завода BC 7 скриптом не меняется (модуль MineTweaker для BC - только под BC 6.0),
// поэтому побочные продукты делаются из нефти и топлива в плотнике Forestry.
// Баланс сырой нефти в ДВС - через config/buildcraft/main.cfg (см. README).
// ====================================================================================

// --- Пековое углеволокно: угольная пыль, пропитанная нефтяным пеком ---
// БЫЛО только 4 угольной пыли (= кокса) на 1 волокно. ТЕПЕРЬ второй путь: 2 пыли + 250 mB нефти -> 2 волокна.
mods.forestry.Carpenter.addRecipe(<IC2:itemPartCarbonFibre> * 2, [
    [<IC2:itemDust:2>, null, <IC2:itemDust:2>]
], <liquid:oil> * 250, 40);

// --- Тротил: толуол (лёгкая фракция = топливо) + нитрование селитрой и серной кислотой (сера) ---
// Нитрование толуола - с охлаждением: 1 лёд (натуральный или из Glacial Precipitator TE).
mods.forestry.Carpenter.addRecipe(<minecraft:tnt> * 2, [
    [<Railcraft:dust:2>, <ore:dustSulfur>, <Railcraft:dust:2>],
    [<minecraft:sand>,   <minecraft:sand>, <minecraft:sand>],
    [null,               <minecraft:ice>,  null]
], <liquid:fuel> * 250, 60);

// --- Синтетический каучук: бутадиен-стирольный (СКС) из лёгкой фракции, вулканизация серой ---
// Бутадиен и стирол получают крекингом бензиновой фракции (топливо), эмульсионная полимеризация.
// Натуральный путь (латекс + сера в индукционной плавильне) остаётся; этот - промышленный, на потоке.
mods.forestry.Carpenter.addRecipe(<IC2:itemRubber> * 4, [
    [<ore:dustSulfur>, null, <ore:dustSulfur>]
], <liquid:fuel> * 500, 60);

// --- Синтетический каучук: бутадиен из крекинга нефти + вулканизация серой ---
// Транспозер TE как реактор полимеризации. 1 сера + 500 mB нефти -> 2 резины.
// Дороже латекса (500 mB нефти = 750 000 RF в виде топлива) - путь для тех, у кого нет плантаций.
mods.thermalexpansion.Transposer.addFillRecipe(4000, <IC2:itemDust:13>, <IC2:itemRubber> * 2, <liquid:oil> * 500);

// --- Синтетическое волокно (акрил): пропилен из нефти + азот из селитры -> 8 нитей (= 2 шерсти) ---
mods.forestry.Carpenter.addRecipe(<minecraft:string> * 8, [
    [<Railcraft:dust:2>]
], <liquid:oil> * 500, 60);

// --- Синтетическая нефть из угля (газификация паром + синтез Фишера-Тропша, как у Sasol) ---
// Уголь + пар -> синтез-газ -> жидкие углеводороды. Консервирующая машина IC2 как реактор.
// Жёсткий выход: 8 угля + 1000 mB пара -> 200 mB нефти (с 1 угля ~25 mB). Нефть дальше - в перегонку.
mods.ic2.Canner.addEnrichRecipe(<liquid:oil> * 200, <liquid:steam> * 1000, <minecraft:coal> * 8);
mods.ic2.Canner.addEnrichRecipe(<liquid:oil> * 200, <liquid:ic2steam> * 1000, <minecraft:coal> * 8);

// Синтетический каучук по Лебедеву (СССР, 1932): спирт -> бутадиен -> каучук. 1000 mB биоэтанола + сера -> 2 резины.
mods.thermalexpansion.Transposer.addFillRecipe(4000, <IC2:itemDust:13>, <IC2:itemRubber> * 2, <liquid:bioethanol> * 1000);

// ====================================================================================
// ПОРОХ
// Дымный (селитра + сера + древесный уголь, верстак): 2 селитры + сера + уголь -> 1 порох (БЫЛО у Railcraft 2).
// Бездымный (нитроцеллюлоза, плотник) - woodworking.zs: те же 2 селитры дают 4 пороха.
// Удалён скрытый рецепт IC2 "редстоун + угольная пыль -> 3 пороха" (порох без селитры).
// Извлечение серы из пороха в экстракторе отключено (config/ic2/extractor.ini).
// ====================================================================================
recipes.remove(<minecraft:gunpowder>);
recipes.addShapeless(<minecraft:gunpowder>, [<Railcraft:dust:2>, <Railcraft:dust:2>, <ore:dustSulfur>, <ore:dustCharcoal>]);
