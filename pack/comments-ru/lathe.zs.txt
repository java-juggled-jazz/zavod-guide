// ====================================================================================
// ТОКАРНЫЙ СТАНОК IC2 (Turning Table, blockMachine3:8) - ЦЕХ ВАЛОВ, ВТУЛОК И КОРОНОК
// ПОЧЕМУ: станок был нужен для двух-трёх рецептов. А вал, коленвал и буровая коронка - это токарка.
// КАК: железная заготовка (2 слитка + железная оболочка) ставится в станок, резец снимает металл
//   по 5 сегментам. Профиль = радиусы сегментов от 5 до 1. Станок питается кинетикой (kU).
//   Профиль хранится в NBT заготовки (state: l0..l4). Несимметричные коронки - в любую сторону.
// Профили сборки:
//   3-3-3-3-3  вал                -> железный вал IC2 (дальше - цементация в сталь)
//   4-2-4-2-4  коленвал         -> ДВС BC, торфяной и биогазовый двигатели Forestry,
//                                             коммерческий и промышленный паровые двигатели Railcraft
//   2-2-2-2-2  оправка (не расходуется) -> 4 втулки из 4 оловянных пластин
//   4-4-3-2-1  буровая коронка    -> бур IC2
//   4-3-2-1-1  алмазная коронка   -> алмазный бур (lategame.zs)
// Ранняя игра не блокируется: втулки вручную (1 из 4 пластин), вал из блока железа в металлоформовщике.
// ====================================================================================

// Вал: точёная заготовка -> железный вал IC2 (БЫЛО только блок железа в металлоформовщике = 9 слитков)
recipes.addShapeless(<IC2:itemRecipePart:11>, [<IC2:itemTurningBlanks>.onlyWithTag({state: {l0: 3, l1: 3, l2: 3, l3: 3, l4: 3}})]);
// Стальной вал: железный вал -> доменная печь IC2 (цементация), config/ic2/blast_furnace.ini

// Бур IC2: БЫЛО 6 железных пластин + силовой блок. ТЕПЕРЬ точёная коронка + железный редуктор.
recipes.remove(<IC2:itemToolDrill>);
recipes.addShaped(<IC2:itemToolDrill>, [
    [null,            (<IC2:itemTurningBlanks>.onlyWithTag({state: {l0: 4, l1: 4, l2: 3, l3: 2, l4: 1}}) | <IC2:itemTurningBlanks>.onlyWithTag({state: {l0: 1, l1: 2, l2: 3, l3: 4, l4: 4}})), null],
    [<ore:plateIron>, <ore:gearIron>,                 <ore:plateIron>],
    [<ore:plateIron>, <IC2:itemRecipePart:2>,         <ore:plateIron>]
]);
